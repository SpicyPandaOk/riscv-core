main:
    li x2, 1020 # stack pointer
    li x10, 5 # n
    call sum_to

halt2:
    j halt2

sum_to:
    beqz x10, base_case
    addi x2, x2, -8 # dec sp
    sw x1, 4(x2) #this frames ra
    sw x10, 0(x2) # this frames n
    addi x10, x10, -1 # dec n
    call sum_to
    lw x11, 0(x2) #load n
    lw x1, 4(x2) #load sp
    addi x2, x2, 8 # pop from sp
    add x10, x10, x11  # add to sum
    mv x3, x10
    ret

base_case:
    ret
